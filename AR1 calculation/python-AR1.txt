def AR_func(a1, a2):
    std_a1 = np.std(a1)
    std_a2 = np.std(a2)
    if std_a1 == 0 or std_a2 == 0:
        return 0
    value = np.mean(((a1 - np.mean(a1)) / std_a1) * ((a2 - np.mean(a2)) / std_a2))
    return value


def CalAR1(var, length):
    AR1_result = []
    for i in range(len(var) - length):
        a1 = var[i: i + length]
        a2 = var[i + 1: i + length + 1]
        AR1_result.append(AR_func(a1, a2))
    return AR1_result


def CalVarn(var, length):
    Varn_result = []
    for i in range(0, len(var) - length + 1, 1):
        a1 = var[i:i + length]
        Varn_result.append(np.var(a1))
    return Varn_result


def PlotOriDat(ax, time1, val1, tp, quanname):
    indices = np.linspace(0, len(time1) - 1, 5, dtype=int)
    ax.plot(time1[:tp + 1], val1[:tp + 1], linestyle='-', linewidth=1.5, color=color1_line)
    ax.plot(time1[tp:], val1[tp:], linestyle='-', linewidth=1.5, color=color2_line)
    ymin = np.min(val1)
    ymax = np.max(val1)
    yM = ymax + (ymax - ymin) * 0.1
    ym = ymin - (ymax - ymin) * 0.1
    ax.set_ylim([ym, yM])
    ax.set_xlim([time1[0], time1[-1]])
    ax.fill_between([time1[0], time1[tp]], ym, yM, alpha=0.15, color=color1_fill, edgecolor=None)
    ax.fill_between([time1[tp], time1[-1]], ym, yM, alpha=0.2, color=color2_fill, edgecolor=None)
    ax.set_xticks([time1[i] for i in indices])
    ax.set_xticklabels([time1[i] for i in indices])
    ax.set_ylabel(quanname)
    return
